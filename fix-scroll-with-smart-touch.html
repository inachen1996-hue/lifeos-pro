🔧 滚动修复 - 智能触摸事件处理

🎯 修复策略

核心思路：
• 使用 useRef 和 useEffect 手动管理事件监听器
• 设置 { passive: false } 选项
• 智能判断滑动方向，只在需要时阻止默认行为

📋 完整修复代码

将以下代码应用到 index.html 的 TimerPage 组件中：

// 在 TimerPage 组件开头添加 useRef const timerCardsRef = useRef([]); // 修改触摸事件处理逻辑 const [touchStart, setTouchStart] = useState(null); const [touchEnd, setTouchEnd] = useState(null); // 智能触摸事件处理函数 const handleTouchStart = useCallback((e, timerId) => { const touch = e.targetTouches[0]; setTouchStart({ x: touch.clientX, y: touch.clientY, time: Date.now(), timerId }); setTouchEnd(null); }, []); const handleTouchMove = useCallback((e, timerId) => { if (!touchStart || touchStart.timerId !== timerId) return; const touch = e.targetTouches[0]; const deltaX = Math.abs(touch.clientX - touchStart.x); const deltaY = Math.abs(touch.clientY - touchStart.y); // 判断是否为水平滑动 const isHorizontalSwipe = deltaX > deltaY && deltaX > 10; // 只在水平滑动时阻止默认行为 if (isHorizontalSwipe && e.cancelable) { e.preventDefault(); } setTouchEnd({ x: touch.clientX, y: touch.clientY }); }, [touchStart]); const handleTouchEnd = useCallback((timerId) => { if (!touchStart || !touchEnd || touchStart.timerId !== timerId) return; const distance = touchStart.x - touchEnd.x; const verticalDistance = Math.abs(touchStart.y - touchEnd.y); const isLeftSwipe = distance > 50 && verticalDistance < 100; if (isLeftSwipe) { setSwipedTimerId(timerId); } setTouchStart(null); setTouchEnd(null); }, [touchStart, touchEnd]); // 使用 useEffect 手动添加事件监听器 useEffect(() => { const cards = timerCardsRef.current; const addTouchListeners = (card, timerId) => { if (!card) return; const touchStartHandler = (e) => handleTouchStart(e, timerId); const touchMoveHandler = (e) => handleTouchMove(e, timerId); const touchEndHandler = () => handleTouchEnd(timerId); // 添加事件监听器，设置 passive: false card.addEventListener('touchstart', touchStartHandler, { passive: false }); card.addEventListener('touchmove', touchMoveHandler, { passive: false }); card.addEventListener('touchend', touchEndHandler); // 返回清理函数 return () => { card.removeEventListener('touchstart', touchStartHandler); card.removeEventListener('touchmove', touchMoveHandler); card.removeEventListener('touchend', touchEndHandler); }; }; // 为所有计时器卡片添加事件监听器 const cleanupFunctions = []; timers.forEach((timer, index) => { const card = cards[index]; if (card) { const cleanup = addTouchListeners(card, timer.id); if (cleanup) cleanupFunctions.push(cleanup); } }); // 清理函数 return () => { cleanupFunctions.forEach(cleanup => cleanup()); }; }, [timers, handleTouchStart, handleTouchMove, handleTouchEnd]);

🔄 JSX 修改

在计时器卡片的 JSX 中：

// 移除原有的 onTouchStart, onTouchMove, onTouchEnd // 添加 ref 引用 <div ref={el => timerCardsRef.current[index] = el} className="timer-card macaron-card bg-white p-6 rounded-2xl border-2 shadow-airy-soft transition-all relative" style={{ borderColor: isRunning ? categoryColor : '#E5E7EB', WebkitTapHighlightColor: 'transparent' }} onClick={() => isSwiped && setSwipedTimerId(null)} onContextMenu={(e) => e.preventDefault()} > {/* 卡片内容 */} </div>

🎯 计划页面修复

对于计划页面的拖拽功能，使用类似的方法：

// 在计划页面组件中添加 const planItemsRef = useRef([]); // 修改触摸拖拽处理 const handlePlanTouchStart = useCallback((e, index) => { if (!isMobile) return; const touch = e.touches[0]; setDraggedIndex(index); setTouchDragState({ isDragging: true, startY: touch.clientY, startX: touch.clientX, currentY: touch.clientY, currentX: touch.clientX }); }, [isMobile]); const handlePlanTouchMove = useCallback((e, index) => { if (!isMobile || !touchDragState.isDragging) return; const touch = e.touches[0]; const currentY = touch.clientY; const currentX = touch.clientX; const deltaY = Math.abs(currentY - touchDragState.startY); const deltaX = Math.abs(currentX - touchDragState.startX); // 判断是否为垂直拖拽 const isVerticalDrag = deltaY > deltaX && deltaY > 10; // 只在垂直拖拽时阻止默认行为 if (isVerticalDrag && e.cancelable) { e.preventDefault(); } setTouchDragState(prev => ({ ...prev, currentY, currentX })); }, [isMobile, touchDragState]); // 使用 useEffect 添加计划项目的事件监听器 useEffect(() => { if (!isMobile) return; const items = planItemsRef.current; const cleanupFunctions = []; planItems.forEach((item, index) => { const element = items[index]; if (!element) return; const touchStartHandler = (e) => handlePlanTouchStart(e, index); const touchMoveHandler = (e) => handlePlanTouchMove(e, index); element.addEventListener('touchstart', touchStartHandler, { passive: false }); element.addEventListener('touchmove', touchMoveHandler, { passive: false }); cleanupFunctions.push(() => { element.removeEventListener('touchstart', touchStartHandler); element.removeEventListener('touchmove', touchMoveHandler); }); }); return () => { cleanupFunctions.forEach(cleanup => cleanup()); }; }, [planItems, isMobile, handlePlanTouchStart, handlePlanTouchMove]);
